 graph TD
    A[Товарная нефть] --> B(Атмосферная перегонка нефти);

    subgraph Основные продукты перегонки
        B --> C[Мазут];
        B --> D[Бензиновая фракция];
        B --> E[Керосиновая фракция];
        B --> F[Дизельная (газойль) фракция];
        B --> G[Лёгкие газы C3-C4];
    end

    %% 1. Газы
    G --> L(ГФУ);
    L --> L1[Сжиженные газы];

    %% 2. Бензин
    D --> J(Каталитический реформинг);
    J --> J1[Высокооктановые компоненты товарного бензина];

    %% 3. Средние дистилляты
    subgraph Гидроочистка
        E --> K;
        F --> K(Установка гидроочистки);
    end
    K --> K1[Реактивное топливо];
    K --> K2[Дизельное топливо];

    %% 4. Вакуумная перегонка
    C --> H(Вакуумная перегонка нефти);
    H --> H1[Лёгкие масляные фракции];
    H --> H2[Средние масляные фракции];
    H --> H3[Тяжёлые масляные фракции];
    H --> I[Гудрон (остаток)];

    %% 5. Производство базовых масел и мазута
    I --> M(Деасфальтизация пропаном);
    M --> N[ДАО];

    N --> O(Селективная очистка);
    O --> O1[Рафинат];
    O --> O2[Экстракт];

    %% 5a. Производство масел
    O1 --> P(Депарафинизация);
    P --> P1[Деваксед-масло];
    P --> P2[Гач];

    P1 --> S(Гидроочистка);
    S --> S1[Базовое масло];

    S1 --> T(Компаундирование)
    T -- Присадки --> T;
    T --> T1[Товарное смазочное масло];

    %% 5b. Производство товарного мазута
    O2 --> Q(Смешение);
    I --> Q;
    Q --> R[Товарный мазут];

    %% Стилизация
    classDef product fill:#ccf, stroke:#333, stroke-width:2px;
    classDef process fill:#f9f, stroke:#333;
    class J1, K1, K2, L1, T1, R product;
    class B, H, J, K, L, M, O, P, S, T, Q process;
